SELECT CONCAT(
    'https://api.aladhan.com/v1/calendar/2025/06?latitude=',
    prayer_times_cities.latitude,
    '&longitude=',
    prayer_times_cities.longitude,
    '&method=4'
) AS url,
    prayer_times.city_id, 
    MIN(prayer_times.fajr) AS fajr, 
    prayer_times_cities.city_name, 
    prayer_times_cities.state_name,
    prayer_times_cities.latitude, 
    prayer_times_cities.longitude
    
FROM 
    prayer_times
JOIN 
    prayer_times_cities ON prayer_times.city_id = prayer_times_cities.id
GROUP BY 
    prayer_times.city_id,
    prayer_times_cities.city_name, 
    prayer_times_cities.state_name,
    prayer_times_cities.latitude, 
    prayer_times_cities.longitude
ORDER BY 
    fajr ASC;

SELECT 
  tp.id,
  tp.program_name,
  tp.channel_id,
  tp.program_image AS old_image,
  tsd.program_image AS new_image,
  tp.program_type AS old_type,
  tsd.program_type AS new_type
FROM tv_programs tp
JOIN tv_channels_schedule_details tsd
  ON tp.channel_id = tsd.channel_id
 AND tp.program_name = tsd.program_name;

UPDATE tv_programs tp
JOIN tv_channels_schedule_details tsd
  ON tp.channel_id = tsd.channel_id
 AND tp.program_name = tsd.program_name
SET 
  tp.program_image = tsd.program_image,
  tp.program_type = tsd.program_type;
